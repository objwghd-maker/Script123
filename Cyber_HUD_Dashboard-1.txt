local Registry = {
    ThemedBackground = {},

local function register(bucket, inst, prop)
    table.insert(Registry[bucket], {inst, prop})
end

local function corner(inst, radius)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, radius or 8)
    c.Parent = inst
    return c
end

local function stroke(inst, color, thickness, transparency)
    local s = Instance.new("UIStroke")
    s.Color = color
    s.Thickness = thickness or 1
    s.Transparency = transparency or 0.5
    s.Parent = inst
    return s
end

local function tween(inst, props, time, style, direction)
    local t = TweenService:Create(inst, TweenInfo.new(time or 0.22, style or Enum.EasingStyle.Quad, direction or Enum.EasingDirection.Out), props)
    t:Play()
    return t
end

local function playSound(id, volume)
    if not State.SoundEnabled then return end
    local ok, snd = pcall(function()
        local s = Instance.new("Sound")
        s.SoundId = id
        s.Volume = volume or 0.4
        s.Parent = SoundService
        return s
    end)
    if ok and snd then
        snd:Play()
        Debris:AddItem(snd, 2)
    end
end

local function makeBarGraph(parent, sampleCount, height)
    local holder = Instance.new("Frame")
    holder.Parent = parent
    holder.Size = UDim2.new(1, 0, 0, height)
    register("ThemedPanel", holder, "BackgroundColor3")
    corner(holder, 8)

    local barsHolder = Instance.new("Frame")
    barsHolder.Parent = holder
    barsHolder.BackgroundTransparency = 1
    barsHolder.Position = UDim2.new(0, 6, 0, 6)
    barsHolder.Size = UDim2.new(1, -12, 1, -12)

    local layout = Instance.new("UIListLayout")
    layout.FillDirection = Enum.FillDirection.Horizontal
    layout.VerticalAlignment = Enum.VerticalAlignment.Bottom
    layout.Padding = UDim.new(0, 2)
    layout.Parent = barsHolder

    local bars = {}
    for i = 1, sampleCount do
        local bar = Instance.new("Frame")
        bar.LayoutOrder = i
        bar.Parent = barsHolder
        bar.AnchorPoint = Vector2.new(0, 1)
        bar.Position = UDim2.new(0, 0, 1, 0)
        bar.Size = UDim2.new(0, math.floor(300 / sampleCount), 0, 2)
        register("ThemedAccentText", bar, "BackgroundColor3")
        corner(bar, 2)
        bars[i] = bar
    end

    return holder, bars
end

local function updateBarGraph(bars, values, maxValue)
    for i, bar in ipairs(bars) do
        local v = values[i] or 0
        local pct = maxValue > 0 and math.clamp(v / maxValue, 0.02, 1) or 0.02
        bar.Size = UDim2.new(0, bar.Size.X.Offset, pct, 0)
    end
end

local function notify(text, kind)

    local existing = ToastHolder:GetChildren()
    local count = 0
    for _, c in ipairs(existing) do
        if c:IsA("Frame") then count += 1 end
    end

        for _, c in ipairs(existing) do
            if c:IsA("Frame") then
                c:Destroy()

            end
        end
    end

    lbl.TextColor3 = theme.TextPrimary
    lbl.TextSize = 12
    lbl.TextTransparency = 1
    lbl.TextXAlignment = Enum.TextXAlignment.Left

local currentPage = nil
local function selectTab(name)
    if currentPage == name then return end
    currentPage = name
    playSound(UiSwooshSoundId, 0.25)
    local theme = Themes[State.ThemeName]
    for tabName, btn in pairs(tabButtons) do
        local active = tabName == name
        btn.TextColor3 = active and theme.TextPrimary or theme.TextMuted
        local indicator = btn:FindFirstChild("Indicator")
        tween(indicator, {BackgroundTransparency = active and 0 or 1}, 0.15)
    end

    end

function applyTheme(name)
    local theme = Themes[name]
    if not theme then return end
    State.ThemeName = name
    for _, entry in ipairs(Registry.ThemedBackground) do
        tween(entry[1], {[entry[2]] = theme.Background}, 0.2)
    end
    for _, entry in ipairs(Registry.ThemedPanel) do
        tween(entry[1], {[entry[2]] = theme.Panel}, 0.2)
    end
    for _, entry in ipairs(Registry.ThemedSidebar) do
        tween(entry[1], {[entry[2]] = theme.Sidebar}, 0.2)
    end
    for _, entry in ipairs(Registry.ThemedMutedText) do
        entry[1][entry[2]] = theme.TextMuted
    end
    for _, entry in ipairs(Registry.ThemedPrimaryText) do
        entry[1][entry[2]] = theme.TextPrimary
    end
    for _, entry in ipairs(Registry.ThemedAccentText) do
        entry[1][entry[2]] = theme.Accent
    end
    AccentGradient.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0, theme.Accent),
        ColorSequenceKeypoint.new(0.5, theme.AccentAlt),
        ColorSequenceKeypoint.new(1, theme.Accent),
    })
    GlowStroke.Color = theme.Accent
    HealthValue.TextColor3 = theme.Good
    PingValue.TextColor3 = theme.Good
    for _, btn in pairs(tabButtons) do
        if btn.Name == currentPage then
            btn.TextColor3 = theme.TextPrimary
        end
    end
end

local frameCount, fpsAccum = 0, 0
local lastPing = 0
local pingHistoryForJitter = {}

local function healthColor(pct)
    local theme = Themes[State.ThemeName]
    if pct > 0.5 then return theme.Good
    elseif pct > 0.2 then return theme.Warn
    else return theme.Bad end
end

local function pingColor(ms)
    local theme = Themes[State.ThemeName]
    if ms < 100 then return theme.Good
    elseif ms < 250 then return theme.Warn
    else return theme.Bad end
end

local humanoidStateNames = {}
for _, e in ipairs(Enum.HumanoidStateType:GetEnumItems()) do
    humanoidStateNames[e] = e.Name
end

RunService.Heartbeat:Connect(function(dt)
    frameCount += 1
    fpsAccum += dt
    if fpsAccum >= 0.5 then
        local fps = frameCount / fpsAccum
        if State.ShowFps then
            FpsValue.Text = tostring(math.floor(fps))
        end
        table.insert(frameTimeSamples, (fpsAccum / frameCount) * 1000)
        table.remove(frameTimeSamples, 1)
        updateBarGraph(FrameTimeBars, frameTimeSamples, 50)
        frameCount, fpsAccum = 0, 0
    end

    local char = LocalPlayer.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        local root = char:FindFirstChild("HumanoidRootPart")
        if hum then
            HealthValue.Text = math.floor(hum.Health) .. " / " .. math.floor(hum.MaxHealth)
            HealthValue.TextColor3 = healthColor(hum.Health / math.max(hum.MaxHealth, 1))
            SpeedValue.Text = tostring(math.floor(hum.WalkSpeed))
            JumpValue.Text = tostring(math.floor(hum.JumpPower))
            local ok, stateName = pcall(function() return humanoidStateNames[hum:GetState()] end)
            StateValue.Text = ok and stateName or "Unknown"
        end
        if root then
            VelocityValue.Text = tostring(math.floor(root.AssemblyLinearVelocity.Magnitude))
        end
        GravityValue.Text = tostring(math.floor(workspace.Gravity))
    end
end)

task.spawn(function()
    while ScreenGui.Parent do
        local ms = math.floor(LocalPlayer:GetNetworkPing() * 1000)
        PingValue.Text = ms .. " ms"
        PingValue.TextColor3 = pingColor(ms)
        StatusDot.BackgroundColor3 = pingColor(ms)

        table.insert(pingSamples, ms)
        table.remove(pingSamples, 1)
        updateBarGraph(PingBars, pingSamples, 300)

        table.insert(pingHistoryForJitter, ms)
        if #pingHistoryForJitter > 10 then
            table.remove(pingHistoryForJitter, 1)
        end
        if #pingHistoryForJitter >= 4 then
            local sum, count = 0, 0
            for i = 2, #pingHistoryForJitter do
                sum += math.abs(pingHistoryForJitter[i] - pingHistoryForJitter[i - 1])
                count += 1
            end
            local avgJitter = sum / math.max(count, 1)
            if avgJitter < 15 then
                ConnectionQualityValue.Text = "Stable"
                ConnectionQualityValue.TextColor3 = Themes[State.ThemeName].Good
            elseif avgJitter < 40 then
                ConnectionQualityValue.Text = "Fluctuating"
                ConnectionQualityValue.TextColor3 = Themes[State.ThemeName].Warn
            else
                ConnectionQualityValue.Text = "Unstable"
                ConnectionQualityValue.TextColor3 = Themes[State.ThemeName].Bad
                pushLog("Lag spike detected: " .. ms .. "ms")
            end
        end

        local allPlayers = Players:GetPlayers()
        PlayerCountValue.Text = #allPlayers .. " / " .. Players.MaxPlayers
        local ageSum = 0
        for _, plr in ipairs(allPlayers) do
            ageSum += plr.AccountAge
        end
        if #allPlayers > 0 then
            AvgAccountAgeValue.Text = math.floor(ageSum / #allPlayers) .. " days"
        end

        SessionUptimeValue.Text = math.floor(os.clock() - State.SessionStart) .. "s"

        for _, def in ipairs(memoryTags) do
            local ok, mb = pcall(function()
                if def.tag == nil then
                    return Stats:GetTotalMemoryUsageMb()
                else
                    return Stats:GetMemoryUsageMbForTag(def.tag)
                end
            end)
            if ok and memoryRows[def.label] then
                memoryRows[def.label].Text = string.format("%.1f MB", mb)
            end
        end

        task.wait(1)
    end
end)
