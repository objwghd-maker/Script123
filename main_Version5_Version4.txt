local function register(bucket, inst, prop)
    table.insert(Registry[bucket], {inst, prop})
end

local function corner(inst, radius)
    local base = radius or 8
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, base)
    c.Parent = inst
    table.insert(Registry.ScalableCorners, {c, base})
    return c
end

local function smartText(label, minSize)
    local maxSize = label.TextSize
    if not maxSize or maxSize <= 0 then return label end
    label.TextScaled = true
    local constraint = Instance.new("UITextSizeConstraint")
    constraint.MaxTextSize = maxSize
    constraint.MinTextSize = minSize or 6
    constraint.Parent = label
    table.insert(Registry.ScalableTextConstraints, {constraint, maxSize, minSize or 6})
    return label
end

local function stroke(inst, color, thickness, transparency)
    local base = thickness or 1
    local s = Instance.new("UIStroke")
    s.Color = color
    s.Thickness = base
    s.Transparency = transparency or 0.5
    s.Parent = inst
    table.insert(Registry.ScalableStrokes, {s, base})
    return s
end

local function tween(inst, props, time, style, direction)
    local t = TweenService:Create(inst, TweenInfo.new(time or 0.22, style or Enum.EasingStyle.Quad, direction or Enum.EasingDirection.Out), props)
    t:Play()
    return t
end

local function playSound(id, volume)
    if not State.SoundEnabled then return end
    local ok, snd = pcall(function()
        local s = Instance.new("Sound")
        s.SoundId = id
        s.Volume = volume or 0.4
        s.Parent = SoundService
        return s
    end)
    if ok and snd then
        snd:Play()
        Debris:AddItem(snd, 2)
    end
end

                local ok2, response = pcall(httpGet, {Url = logoUrl, Method = "GET"})

                        writefile("dashboard_logo_cache.png", response.Body)
                        finalImage = getcustomasset("dashboard_logo_cache.png")

        end
    end
    if loaded and finalImage then

        ReopenBubble.Image = finalImage

local function updateBarGraph(bars, values, maxValue)
    for i, bar in ipairs(bars) do

        bar.Size = UDim2.new(0, bar.Size.X.Offset, pct, 0)
    end
end

        if c:IsA("Frame") then count = count + 1 end

        for _, c in ipairs(existing) do
            if c:IsA("Frame") then

            end
        end
    end

    local toast = Instance.new("Frame")

    toast.Size = UDim2.new(1, 0, 0, 36)

    local lbl = Instance.new("TextLabel")

    lbl.BackgroundTransparency = 1
    lbl.Size = UDim2.new(1, -16, 1, 0)
    lbl.Position = UDim2.new(0, 8, 0, 0)
    lbl.Font = Enum.Font.GothamMedium
    lbl.Text = text
    lbl.TextColor3 = theme.TextPrimary
    lbl.TextSize = 12
    lbl.TextTransparency = 1
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    smartText(lbl, 8)

    tween(toast, {BackgroundTransparency = 0.1}, 0.2)
    tween(lbl, {TextTransparency = 0}, 0.2)
    task.delay(2.5, function()
        tween(toast, {BackgroundTransparency = 1}, 0.3)
        tween(lbl, {TextTransparency = 1}, 0.3)
        task.wait(0.3)
        toast:Destroy()
    end)

local currentPage = nil
local function selectTab(name)
    if currentPage == name then return end
    currentPage = name
    playSound(UiSwooshSoundId, 0.25)
    local theme = Themes[State.ThemeName]
    for tabName, data in pairs(tabButtons) do
        local active = tabName == name
        data.titleLbl.TextColor3 = active and theme.TextPrimary or theme.TextMuted
        local indicator = data.button:FindFirstChild("Indicator")
        tween(indicator, {BackgroundTransparency = active and 0 or 1}, 0.15)
        tween(data.button, {BackgroundTransparency = active and 0.9 or 1}, 0.15)
    end
    for pageName, frame in pairs(pageFrames) do
        if pageName == name then
            frame.Visible = true
            frame.Position = UDim2.new(0, 12, 0, 0)
            tween(frame, {Position = UDim2.new(0, 0, 0, 0)}, 0.3, Enum.EasingStyle.Back)
        else
            frame.Visible = false
        end
    end
end

function applyTheme(name)
    local theme = Themes[name]
    if not theme then return end
    State.ThemeName = name
    for _, entry in ipairs(Registry.ThemedBackground) do
        tween(entry[1], {[entry[2]] = theme.Background}, 0.2)
    end
    for _, entry in ipairs(Registry.ThemedPanel) do
        tween(entry[1], {[entry[2]] = theme.Panel}, 0.2)
    end
    for _, entry in ipairs(Registry.ThemedCard) do
        tween(entry[1], {[entry[2]] = theme.Card}, 0.2)
    end
    for _, entry in ipairs(Registry.ThemedSidebar) do
        tween(entry[1], {[entry[2]] = theme.Sidebar}, 0.2)
    end
    for _, entry in ipairs(Registry.ThemedMutedText) do
        entry[1][entry[2]] = theme.TextMuted
    end
    for _, entry in ipairs(Registry.ThemedPrimaryText) do
        entry[1][entry[2]] = theme.TextPrimary
    end
    for _, entry in ipairs(Registry.ThemedAccentText) do
        entry[1][entry[2]] = theme.Accent
    end
    GlowStroke.Color = Color3.fromRGB(255, 145, 0)
    HealthValue.TextColor3 = theme.Good
    PingValue.TextColor3 = theme.Good
    for tabName, data in pairs(tabButtons) do
        if tabName == currentPage then
            data.titleLbl.TextColor3 = theme.TextPrimary
        end
    end
end

local ResponsiveScale = Instance.new("UIScale")
ResponsiveScale.Parent = MainFrame

local BaseWidth, BaseHeight = 680, 440
local MinScale = 0.35

local function getViewportBounds()
    local camera = workspace.CurrentCamera
    if not camera then
        return Vector2.new(0, 0)
    end
    return camera.ViewportSize
end

local function computeResponsiveScale()
    local viewport = getViewportBounds()
    if viewport.X <= 0 or viewport.Y <= 0 then
        return 1
    end
    local scale = math.min(viewport.X / BaseWidth, viewport.Y / BaseHeight)
    return math.clamp(scale, MinScale, 1)
end

local function clampMainFramePosition()
    local viewport = getViewportBounds()
    if viewport.X <= 0 or viewport.Y <= 0 then return end

    local scale = math.max(ResponsiveScale.Scale, 0.01)
    local halfW = MainFrame.Size.X.Offset * scale * 0.5
    local halfH = MainFrame.Size.Y.Offset * scale * 0.5
    local margin = 6

    local minX = -viewport.X * 0.5 + halfW + margin
    local maxX = viewport.X * 0.5 - halfW - margin
    local minY = -viewport.Y * 0.5 + halfH + margin
    local maxY = viewport.Y * 0.5 - halfH - margin

    local x = MainFrame.Position.X.Offset
    local y = MainFrame.Position.Y.Offset

    if minX > maxX then
        x = 0
    else
        x = math.clamp(x, minX, maxX)
    end

    if minY > maxY then
        y = 0
    else
        y = math.clamp(y, minY, maxY)
    end

    MainFrame.Position = UDim2.new(0.5, x, 0.5, y)
end

local function adaptLayoutForViewport(scale)
    for _, holder in ipairs(Registry.Columns) do
        local layout = nil
        for _, c in ipairs(holder:GetChildren()) do
            if c:IsA("UIListLayout") then
                layout = c
                break
            end
        end
        if layout then
            if scale < 0.65 then
                layout.FillDirection = Enum.FillDirection.Vertical
                layout.Padding = UDim.new(0, 8)
                for _, child in ipairs(holder:GetChildren()) do
                    if child:IsA("ScrollingFrame") then
                        child.Size = UDim2.new(1, 0, 0, 0)
                        child.AutomaticCanvasSize = Enum.AutomaticSize.Y
                        child.ScrollBarThickness = 3
                        child.CanvasSize = UDim2.new(0, 0, 0, 0)
                    end
                end
            else
                layout.FillDirection = Enum.FillDirection.Horizontal
                layout.Padding = UDim.new(0, 12)
                for _, child in ipairs(holder:GetChildren()) do
                    if child:IsA("ScrollingFrame") then
                        child.Size = UDim2.new(0.5, -6, 1, 0)
                        child.AutomaticCanvasSize = Enum.AutomaticSize.Y
                        child.ScrollBarThickness = 3
                        child.CanvasSize = UDim2.new(0, 0, 0, 0)
                    end
                end
            end
        end
    end
end

local function updateResponsiveScale()
    local targetScale = computeResponsiveScale()
    tween(ResponsiveScale, {Scale = targetScale}, 0.2)
    local scaleFactor = targetScale
    for _, data in ipairs(Registry.ScalableCorners) do
        local c, base = data[1], data[2]
        c.CornerRadius = UDim.new(0, math.max(1, math.floor(base * scaleFactor)))
    end
    for _, data in ipairs(Registry.ScalableStrokes) do
        local s, base = data[1], data[2]
        s.Thickness = math.max(0.5, base * scaleFactor)
    end
    for _, data in ipairs(Registry.ScalableTextConstraints) do
        local constraint, baseMax, baseMin = data[1], data[2], data[3]
        constraint.MaxTextSize = math.max(8, math.floor(baseMax * scaleFactor))
        constraint.MinTextSize = math.max(6, math.floor(baseMin * scaleFactor))
    end
    adaptLayoutForViewport(scaleFactor)
end

local fittingToViewport = false
local function fitFrameToViewport()
    local viewport = getViewportBounds()
    if viewport.X <= 0 or viewport.Y <= 0 then return end

    local margin = 24
    local maxW = math.max(viewport.X - margin * 2, 300)
    local maxH = math.max(viewport.Y - margin * 2, 220)

    local currentW = MainFrame.Size.X.Offset
    local currentH = MainFrame.Size.Y.Offset
    local newW = math.min(currentW, maxW)
    local newH = math.min(currentH, maxH)

    if newW ~= currentW or newH ~= currentH then
        fittingToViewport = true
        MainFrame.Size = UDim2.new(0, newW, 0, newH)
        -- store last known good normal size
        normalSize = MainFrame.Size
        expandedSize = MainFrame.Size
        fittingToViewport = false
    end

    updateResponsiveScale()
    task.defer(clampMainFramePosition)
end

local function hookCamera(camera)
    if not camera then return end
    camera:GetPropertyChangedSignal("ViewportSize"):Connect(fitFrameToViewport)
end

end)

        updateBarGraph(PingBars, pingSamples, 300)

        table.insert(pingHistoryForJitter, ms)
        if #pingHistoryForJitter > 10 then
            table.remove(pingHistoryForJitter, 1)
        end
        if #pingHistoryForJitter >= 4 then
            local sum, count = 0, 0
            for i = 2, #pingHistoryForJitter do
                sum = sum + math.abs(pingHistoryForJitter[i] - pingHistoryForJitter[i - 1])
                count = count + 1
            end
            local avgJitter = sum / math.max(count, 1)
            if avgJitter < 15 then
                ConnectionQualityValue.Text = "Stable"
                ConnectionQualityValue.TextColor3 = Themes[State.ThemeName].Good
            elseif avgJitter < 40 then
                ConnectionQualityValue.Text = "Fluctuating"
                ConnectionQualityValue.TextColor3 = Themes[State.ThemeName].Warn
            else
                ConnectionQualityValue.Text = "Unstable"
                ConnectionQualityValue.TextColor3 = Themes[State.ThemeName].Bad
                pushLog("Lag spike detected: " .. ms .. "ms")
            end
        end

        local allPlayers = Players:GetPlayers()
        PlayerCountValue.Text = #allPlayers .. " / " .. Players.MaxPlayers
        local ageSum = 0
        for _, plr in ipairs(allPlayers) do
            ageSum = ageSum + plr.AccountAge
        end
        if #allPlayers > 0 then
            AvgAccountAgeValue.Text = math.floor(ageSum / #allPlayers) .. " days"
        end

        SessionUptimeValue.Text = math.floor(os.clock() - State.SessionStart) .. "s"

        for _, def in ipairs(memoryTags) do
            local ok, mb = pcall(function()
                if def.tag == nil then
                    return Stats:GetTotalMemoryUsageMb()
                else
                    return Stats:GetMemoryUsageMbForTag(def.tag)
                end
            end)
            if ok and memoryRows[def.label] then
                memoryRows[def.label].Text = string.format("%.1f MB", mb)
            end
        end

        task.wait(1)
    end
end)
